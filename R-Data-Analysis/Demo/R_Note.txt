NewtonRaphson = function(f1, f2, xold = -1, do = TRUE, eps = 1e-6)
{
  start = xold
  while (do)
  {
    x = xold-f1(xold)/f2(xold)
    if ( abs(x-xold) < 1e-6)
    {
      do = FALSE
    }
    xold = x
  }
  return(list(start, xold))
}

# 二分法求根（Bisection Method）
BisectRoot = function(f, a=0, b=2, tol = 1e-6, eps = 1e-7) {
  repeat
  {
    m = (a+b)/2
    if (f(a)*f(m)<0)
    {
      b=m
    }else
    {
      a=m
    }
    do = (abs(b-a) < tol) & (abs(f(m)) < eps)
    if (do)
    {
      break
    }
  }
  return(m)
}
args(BisectRoot)
g = function(x) x^3+2*x^2-7
BisectRoot(g)





# 平方根（DoubleSquared）
c = 3
DoubleSquared = function(c, start = 1, tol = 0.0001) {
  x = start
  do = TRUE
  while (do)
  {
    tmp = 0.5*(x+c/x)
    if(abs(tmp-x)>tol)
    {
      x = tmp
    }else
    {
      do = FALSE
    }
  }
  return(x)
}
